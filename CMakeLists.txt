cmake_minimum_required(VERSION 3.24)
project(udp_vision_receiver VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RECEIVER_GUI_DEFAULT OFF)
if(WIN32)
  set(RECEIVER_GUI_DEFAULT ON)
endif()
option(RECEIVER_GUI "Build the Windows DirectX GUI" ${RECEIVER_GUI_DEFAULT})
option(RECEIVER_TENSORRT "Build the production CUDA/TensorRT backend" OFF)
option(RECEIVER_INTEGRATION_TESTS "Register hardware-free process/network integration tests" OFF)
option(RECEIVER_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
if(RECEIVER_SANITIZERS)
  if(MSVC OR WIN32)
    message(FATAL_ERROR "The combined address/undefined sanitizer preset requires Linux GCC or Clang")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(RECEIVER_CUDA_ARCHITECTURES "86" CACHE STRING "CUDA architectures for the TensorRT backend")
include(FetchContent)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.12.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(json)
find_package(Threads REQUIRED)
add_library(receiver_core src/protocol.cpp src/control.cpp src/config.cpp src/network.cpp src/injection.cpp src/app.cpp src/prediction.cpp src/local_mouse.cpp src/backend.cpp src/yolo_omni.cpp src/engine.cpp src/model.cpp src/preferences.cpp src/performance.cpp)
target_include_directories(receiver_core PUBLIC include)
target_link_libraries(receiver_core PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
if(WIN32)
  target_compile_definitions(receiver_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(receiver_core PUBLIC ws2_32 winmm bcrypt user32)
endif()
if(MSVC)
  target_compile_options(receiver_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/permissive->)
else()
  target_compile_options(receiver_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
endif()
if(RECEIVER_TENSORRT)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(TENSORRT_ROOT "" CACHE PATH "TensorRT 10.x installation root")
  set(TRT_HINTS "${TENSORRT_ROOT}" "$ENV{TENSORRT_ROOT}" /usr /usr/local)
  find_path(TRT_INCLUDE NvInfer.h HINTS ${TRT_HINTS}
            PATH_SUFFIXES include include/x86_64-linux-gnu REQUIRED)
  find_library(TRT_LIB NAMES nvinfer_10 nvinfer HINTS ${TRT_HINTS}
               PATH_SUFFIXES lib lib64 targets/x86_64-linux-gnu/lib REQUIRED)
  find_library(TRT_ONNX NAMES nvonnxparser_10 nvonnxparser HINTS ${TRT_HINTS}
               PATH_SUFFIXES lib lib64 targets/x86_64-linux-gnu/lib REQUIRED)
  target_sources(receiver_core PRIVATE src/tensorrt.cpp src/preprocess.cu)
  target_include_directories(receiver_core PRIVATE "${TRT_INCLUDE}")
  target_link_libraries(receiver_core PUBLIC CUDA::cudart ${TRT_LIB} ${TRT_ONNX})
  if(NOT WIN32)
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    target_link_libraries(receiver_core PUBLIC OpenSSL::Crypto)
  endif()
  set_target_properties(receiver_core PROPERTIES CUDA_STANDARD 17
                         CUDA_ARCHITECTURES "${RECEIVER_CUDA_ARCHITECTURES}")
  target_compile_definitions(receiver_core PUBLIC RECEIVER_HAS_TENSORRT=1)
else()
  target_sources(receiver_core PRIVATE src/no_backend.cpp)
endif()
add_executable(receiver_headless src/main.cpp)
target_link_libraries(receiver_headless PRIVATE receiver_core)
if(RECEIVER_TENSORRT AND NOT WIN32)
  get_filename_component(TRT_LIBRARY_DIR "${TRT_LIB}" DIRECTORY)
  set_target_properties(receiver_headless PROPERTIES BUILD_RPATH "${TRT_LIBRARY_DIR}"
                        INSTALL_RPATH "${TRT_LIBRARY_DIR}")
endif()
add_executable(receiver_verify src/verify.cpp)
target_link_libraries(receiver_verify PRIVATE receiver_core)
if(RECEIVER_TENSORRT AND NOT WIN32)
  set_target_properties(receiver_verify PROPERTIES BUILD_RPATH "${TRT_LIBRARY_DIR}"
                        INSTALL_RPATH "${TRT_LIBRARY_DIR}")
endif()
install(TARGETS receiver_verify RUNTIME DESTINATION .)
if(RECEIVER_GUI AND WIN32)
  add_library(receiver_testhub src/test_hub.cpp)
  target_include_directories(receiver_testhub PUBLIC include)
  target_compile_definitions(receiver_testhub PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(receiver_testhub PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
  FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.92.1 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(imgui)
  add_library(imgui_lib ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp)
  target_include_directories(imgui_lib PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  target_link_libraries(imgui_lib PUBLIC d3d11 dxgi d3dcompiler dwmapi)
  add_executable(receiver WIN32 src/gui.cpp)
  target_link_libraries(receiver PRIVATE receiver_core imgui_lib comdlg32 receiver_testhub)
  add_executable(receiver_test_hub_runner tests/test_hub_runner.cpp)
  target_link_libraries(receiver_test_hub_runner PRIVATE receiver_testhub)
  install(TARGETS receiver receiver_test_hub_runner RUNTIME DESTINATION .)
endif()
enable_testing()
add_executable(receiver_reliability_tests tests/reliability_tests.cpp)

add_executable(receiver_scenario_runner tests/scenario_runner.cpp)
target_link_libraries(receiver_scenario_runner PRIVATE receiver_core)
add_executable(receiver_inspect_model tests/inspect_model.cpp)
target_link_libraries(receiver_inspect_model PRIVATE receiver_core)
add_executable(receiver_workflow_tests tests/workflow_tests.cpp)
target_link_libraries(receiver_workflow_tests PRIVATE receiver_core)
add_test(NAME receiver_workflows COMMAND receiver_workflow_tests)
set_tests_properties(receiver_workflows PROPERTIES TIMEOUT 30)
add_executable(receiver_backend_tests tests/backend_tests.cpp)
target_link_libraries(receiver_backend_tests PRIVATE receiver_core)
add_test(NAME receiver_backends COMMAND receiver_backend_tests)
find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME receiver_omni_worker COMMAND receiver_backend_tests
           "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/omni_fixture.py")
  set_tests_properties(receiver_omni_worker PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 30)
  add_test(NAME receiver_omni_protocol COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/omni_worker_test.py")
  if(RECEIVER_INTEGRATION_TESTS)
    foreach(pair "network;integration_test.py" "direction;direction_integration_test.py" "recovery;telemetry_recovery_test.py")
      list(GET pair 0 name)
      list(GET pair 1 script)
      add_test(NAME receiver_${name}_pipeline COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/${script}" $<TARGET_FILE:receiver_headless>)
      set_tests_properties(receiver_${name}_pipeline PROPERTIES TIMEOUT 120 RESOURCE_LOCK receiver_loopback LABELS integration)
    endforeach()
    add_test(NAME receiver_model_faults COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/model_fault_integration_test.py" $<TARGET_FILE:receiver_scenario_runner>
             --report "${CMAKE_BINARY_DIR}/test-results/model-faults.json")
    set_tests_properties(receiver_model_faults PROPERTIES TIMEOUT 120 RESOURCE_LOCK receiver_loopback LABELS integration)
    if(RECEIVER_GUI AND WIN32)
      add_test(NAME receiver_gui_pages COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/gui_smoke.py" $<TARGET_FILE:receiver> --all-pages)
      set_tests_properties(receiver_gui_pages PROPERTIES TIMEOUT 120 RESOURCE_LOCK receiver_loopback LABELS integration)
    endif()
  endif()
elseif(RECEIVER_INTEGRATION_TESTS)
  message(FATAL_ERROR "Python is required for the enabled integration tests")
endif()
add_executable(receiver_tests tests/core_tests.cpp)
target_link_libraries(receiver_tests PRIVATE receiver_core)

  add_test(NAME receiver_local_mouse_reader COMMAND receiver_local_mouse_tests)
  set_tests_properties(receiver_local_mouse_reader PROPERTIES TIMEOUT 15 SKIP_RETURN_CODE 77)

target_link_libraries(receiver_decode PRIVATE receiver_core)

install(TARGETS receiver_tests receiver_workflow_tests receiver_reliability_tests receiver_scenario_runner receiver_inspect_model receiver_backend_tests receiver_humanization_tests receiver_tracking_tests receiver_local_mouse_tests receiver_injection_tests receiver_proxy_codec_tests receiver_decode RUNTIME DESTINATION .)
install(DIRECTORY tools docs profiles integrations tests DESTINATION . PATTERN "__pycache__" EXCLUDE)

  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
  include(InstallRequiredSystemLibraries)

  get_filename_component(RECEIVER_COMPILER_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
  foreach(runtime libc++.dll libunwind.dll libwinpthread-1.dll libstdc++-6.dll libgcc_s_seh-1.dll)
    if(EXISTS "${RECEIVER_COMPILER_DIR}/${runtime}")
      install(FILES "${RECEIVER_COMPILER_DIR}/${runtime}" DESTINATION .)
    endif()
  endforeach()
